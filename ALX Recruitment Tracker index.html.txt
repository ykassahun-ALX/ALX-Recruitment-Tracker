<script src="https://cdnjs.cloudflare.com/ajax/libs/PapaParse/5.4.1/papaparse.min.js" crossorigin="anonymous"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/Chart.js/4.5.0/chart.umd.min.js" crossorigin="anonymous"></script>
<script>
(function(){
  "use strict";

  // ---------------- State ----------------
  let learners = [];          // full array of learner objects
  let allPrograms = [];        // sorted list of program names
  let allCountries = [];       // sorted list of country names (by enrollment)
  let allCourses = [];         // sorted list of "Program — Course" strings
  const filters = { programs: new Set(), countries: new Set(), course: "all", search: "" };
  let page = 1;
  const PAGE_SIZE = 50;
  let genderChartRef = null;
  let barChartRef = null;
  let theme = "light"; // kept in memory only (no localStorage inside the artifact preview)

  // ---------------- DOM refs ----------------
  const dropzone = document.getElementById("dropzone");
  const fileInput = document.getElementById("fileInput");
  const fileStatus = document.getElementById("fileStatus");
  const fileStatusText = document.getElementById("fileStatusText");
  const spinner = document.getElementById("spinner");
  const dashboardBody = document.getElementById("dashboardBody");
  const emptyState = document.getElementById("emptyState");
  const topLearnerCount = document.getElementById("topLearnerCount");
  const themeToggle = document.getElementById("themeToggle");
  const themeIcon = document.getElementById("themeIcon");
  const themeLabel = document.getElementById("themeLabel");

  // reads the live CSS custom property values so Chart.js matches the current theme
  function cssVar(name){
    return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
  }
  function chartColors(){
    return { text: cssVar("--muted"), grid: cssVar("--border"), surface: cssVar("--surface") };
  }

  function setTheme(next){
    theme = next;
    document.documentElement.setAttribute("data-theme", theme);
    themeIcon.textContent = theme === "dark" ? "☀️" : "🌙";
    themeLabel.textContent = theme === "dark" ? "Light" : "Dark";
    if (learners.length){
      // charts bake colors into their config at creation time, so redraw them on theme change
      const filtered = getFiltered();
      renderGenderChart(filtered);
      renderBarChart(filtered);
    }
  }
  themeToggle.addEventListener("click", () => setTheme(theme === "dark" ? "light" : "dark"));

  dropzone.addEventListener("click", () => fileInput.click());
  dropzone.addEventListener("dragover", e => { e.preventDefault(); dropzone.classList.add("drag"); });
  dropzone.addEventListener("dragleave", () => dropzone.classList.remove("drag"));
  dropzone.addEventListener("drop", e => {
    e.preventDefault();
    dropzone.classList.remove("drag");
    if (e.dataTransfer.files && e.dataTransfer.files[0]) handleFile(e.dataTransfer.files[0]);
  });
  fileInput.addEventListener("change", e => {
    if (e.target.files && e.target.files[0]) handleFile(e.target.files[0]);
  });

  function handleFile(file){
    fileStatus.classList.remove("ok");
    spinner.classList.add("show");
    fileStatusText.textContent = "Reading " + file.name + " …";
    // yield a frame first so the "Reading…" message actually paints before the
    // (synchronous) parse work below starts — no Web Worker, since sandboxed
    // artifact iframes can silently block worker creation.
    requestAnimationFrame(() => requestAnimationFrame(() => {
      const reader = new FileReader();
      reader.onload = function(e){
        let parsed;
        try {
          parsed = Papa.parse(e.target.result, { header: true, skipEmptyLines: true });
        } catch (err){
          spinner.classList.remove("show");
          fileStatusText.textContent = "Could not parse that file: " + err;
          return;
        }
        if (parsed.errors && parsed.errors.length){
          console.warn("CSV parse warnings:", parsed.errors.slice(0,5));
        }
        fileStatusText.textContent = "Aggregating " + parsed.data.length.toLocaleString() + " rows…";
        requestAnimationFrame(() => requestAnimationFrame(() => {
          aggregate(parsed.data);
          spinner.classList.remove("show");
          fileStatus.classList.add("ok");
          fileStatusText.textContent = "Loaded " + file.name + " · " + parsed.data.length.toLocaleString() + " rows · " + learners.length.toLocaleString() + " unique learners";
          renderAll(true);
        }));
      };
      reader.onerror = function(){
        spinner.classList.remove("show");
        fileStatusText.textContent = "Could not read that file from disk.";
      };
      reader.readAsText(file);
    }));
  }

  // ---------------- Aggregation ----------------
  function aggregate(rows){
    const map = new Map();
    const courseSet = new Set();

    for (const row of rows){
      const email = (row["Email"] || "").trim().toLowerCase();
      if (!email) continue;

      let L = map.get(email);
      if (!L){
        L = {
          email: email,
          first: row["First name"] || "",
          last: row["Last name"] || "",
          gender: row["Gender"] || "n/a",
          country: row["Country of residence"] || "n/a",
          region: row["Regions"] || "n/a",
          health: row["Learner health classification status"] || "n/a",
          activated: false,
          paid: false,
          programs: new Map() // programName -> {seq, name, status}
        };
        map.set(email, L);
      }
      if (row["Is enrollment activated"] === "Yes") L.activated = true;
      if (row["Payment plan status"] === "ACTIVE") L.paid = true;
      if (row["Learner health classification status"]) L.health = row["Learner health classification status"];

      const program = row["Program name"] || "n/a";
      const courseName = row["Course name"] || "n/a";
      const seqRaw = row["Course sequence number"];
      const seq = (seqRaw === "" || seqRaw === undefined || seqRaw === null) ? -1 : parseInt(seqRaw, 10);
      const status = row["Course status (LMS)"] || "n/a";

      if (courseName !== "n/a" && !isNaN(seq)){
        const existing = L.programs.get(program);
        if (!existing || seq > existing.seq){
          L.programs.set(program, { seq: seq, name: courseName, status: status });
        }
      } else if (!L.programs.has(program)){
        L.programs.set(program, { seq: -1, name: "n/a", status: "n/a" });
      }
    }

    // finalize into plain arrays for fast repeated filtering
    learners = [];
    for (const L of map.values()){
      const progArr = [];
      for (const [name, info] of L.programs.entries()){
        progArr.push({ program: name, course: info.name, status: info.status });
        if (info.name !== "n/a") courseSet.add(name + " — " + info.name);
      }
      learners.push({
        email: L.email, first: L.first, last: L.last, gender: L.gender,
        country: L.country, region: L.region, health: L.health,
        activated: L.activated, paid: L.paid,
        programs: progArr, multi: progArr.length > 1
      });
    }

    // distinct filter option lists
    const progCount = new Map();
    const countryCount = new Map();
    for (const L of learners){
      for (const p of L.programs) progCount.set(p.program, (progCount.get(p.program)||0)+1);
      countryCount.set(L.country, (countryCount.get(L.country)||0)+1);
    }
    allPrograms = [...progCount.keys()].sort((a,b)=>progCount.get(b)-progCount.get(a));
    allCountries = [...countryCount.keys()].sort((a,b)=>countryCount.get(b)-countryCount.get(a));
    allCourses = [...courseSet].sort();

    buildFilterUI();
  }

  // ---------------- Filter UI construction (only rebuilt on new upload) ----------------
  function buildFilterUI(){
    const progWrap = document.getElementById("programChips");
    progWrap.innerHTML = "";
    allPrograms.forEach(p => progWrap.appendChild(makeChip(p, "programs")));

    const countryWrap = document.getElementById("countryChips");
    countryWrap.innerHTML = "";
    allCountries.forEach(c => countryWrap.appendChild(makeChip(c, "countries")));

    const courseSelect = document.getElementById("courseFilter");
    courseSelect.innerHTML = '<option value="all">All modules</option>';
    allCourses.forEach(c => {
      const opt = document.createElement("option");
      opt.value = c; opt.textContent = c;
      courseSelect.appendChild(opt);
    });
  }

  function makeChip(label, setName){
    const el = document.createElement("div");
    el.className = "chip";
    el.textContent = label;
    el.addEventListener("click", () => {
      const set = filters[setName];
      if (set.has(label)) set.delete(label); else set.add(label);
      el.classList.toggle("active");
      page = 1;
      renderAll();
    });
    return el;
  }

  document.getElementById("courseFilter").addEventListener("change", e => {
    filters.course = e.target.value; page = 1; renderAll();
  });
  document.getElementById("searchBox").addEventListener("input", e => {
    filters.search = e.target.value.trim().toLowerCase(); page = 1; renderTable();
  });
  document.getElementById("clearFilters").addEventListener("click", () => {
    filters.programs.clear(); filters.countries.clear(); filters.course = "all"; filters.search = "";
    document.getElementById("searchBox").value = "";
    document.getElementById("courseFilter").value = "all";
    document.querySelectorAll(".chip.active").forEach(c => c.classList.remove("active"));
    page = 1; renderAll();
  });

  document.getElementById("rankMetric").addEventListener("change", () => renderScoreboard(getFiltered()));

  document.getElementById("prevPage").addEventListener("click", () => { if (page>1){ page--; renderTable(); }});
  document.getElementById("nextPage").addEventListener("click", () => { page++; renderTable(); });

  document.getElementById("exportBtn").addEventListener("click", () => {
    const rows = getFiltered();
    const header = ["Email","First name","Last name","Gender","Country","# Programs","Programs","Progress","Health status","Activated","Paid"];
    const lines = [header.join(",")];
    rows.forEach(L => {
      const progs = L.programs.map(p=>p.program).join(" | ");
      const mods = L.programs.map(p=>p.program+": "+p.course+" ("+p.status+")").join(" | ");
      const cells = [L.email, L.first, L.last, L.gender, L.country, L.programs.length, progs, mods, L.health, L.activated?"Yes":"No", L.paid?"Yes":"No"];
      lines.push(cells.map(c => '"' + String(c).replace(/"/g,'""') + '"').join(","));
    });
    const blob = new Blob([lines.join("\n")], { type: "text/csv" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = "learner_outreach_export.csv";
    a.click();
    URL.revokeObjectURL(url);
  });

  // ---------------- Filtering ----------------
  function getFiltered(){
    return learners.filter(L => {
      if (filters.programs.size && !L.programs.some(p => filters.programs.has(p.program))) return false;
      if (filters.countries.size && !filters.countries.has(L.country)) return false;
      if (filters.course !== "all"){
        const match = L.programs.some(p => (p.program + " — " + p.course) === filters.course);
        if (!match) return false;
      }
      return true;
    });
  }
  function getFilteredWithSearch(){
    let rows = getFiltered();
    if (filters.search){
      rows = rows.filter(L =>
        L.email.includes(filters.search) ||
        (L.first + " " + L.last).toLowerCase().includes(filters.search)
      );
    }
    return rows;
  }

  // ---------------- Render orchestration ----------------
  function renderAll(){
    if (!learners.length) return;
    dashboardBody.style.display = "block";
    emptyState.style.display = "none";
    topLearnerCount.textContent = learners.length.toLocaleString();
    const filtered = getFiltered();
    renderStats(filtered);
    renderGenderChart(filtered);
    renderBarChart(filtered);
    renderScoreboard(filtered);
    renderTable();
  }

  function renderStats(rows){
    const enrolled = rows.length;
    const activated = rows.filter(r=>r.activated).length;
    const paid = rows.filter(r=>r.paid).length;
    const multi = rows.filter(r=>r.multi).length;
    document.getElementById("statEnrolled").textContent = enrolled.toLocaleString();
    document.getElementById("statActivated").textContent = activated.toLocaleString();
    document.getElementById("statActivatedPct").textContent = pct(activated,enrolled) + "% of enrolled";
    document.getElementById("statPaid").textContent = paid.toLocaleString();
    document.getElementById("statPaidPct").textContent = pct(paid,enrolled) + "% of enrolled · " + pct(paid,activated) + "% of activated";
    document.getElementById("statMulti").textContent = multi.toLocaleString();
  }

  function pct(n,d){ if (!d) return "0"; return ((n/d)*100).toFixed(1); }

  function renderGenderChart(rows){
    const counts = new Map();
    rows.forEach(r => counts.set(r.gender, (counts.get(r.gender)||0)+1));
    const labels = [...counts.keys()];
    const data = labels.map(l=>counts.get(l));
    const colors = ["#1F6E46","#3B6E9E","#B9762F","#A6483F","#8C8C8C"];
    const c = chartColors();
    if (genderChartRef) genderChartRef.destroy();
    genderChartRef = new Chart(document.getElementById("genderChart"), {
      type: "doughnut",
      data: { labels: labels, datasets: [{ data: data, backgroundColor: colors, borderWidth: 2, borderColor: c.surface }] },
      options: {
        plugins: { legend: { position: "bottom", labels: { boxWidth: 10, font: { size: 11.5 }, color: c.text } } },
        cutout: "62%"
      }
    });
  }

  function renderBarChart(rows){
    const groups = new Map();
    allCountries.forEach(k => groups.set(k, { enrolled:0, activated:0, paid:0 }));

    rows.forEach(L => {
      const g = groups.get(L.country); if (!g) return;
      g.enrolled++; if (L.activated) g.activated++; if (L.paid) g.paid++;
    });

    const labels = [...groups.keys()].filter(k => groups.get(k).enrolled > 0)
      .sort((a,b)=>groups.get(b).enrolled - groups.get(a).enrolled);

    const c = chartColors();
    if (barChartRef) barChartRef.destroy();
    barChartRef = new Chart(document.getElementById("barChart"), {
      type: "bar",
      data: {
        labels: labels,
        datasets: [
          { label:"Enrolled", data: labels.map(l=>groups.get(l).enrolled), backgroundColor: c.grid },
          { label:"Activated", data: labels.map(l=>groups.get(l).activated), backgroundColor:"#3B6E9E" },
          { label:"Paid", data: labels.map(l=>groups.get(l).paid), backgroundColor:"#1F6E46" }
        ]
      },
      options: {
        plugins: { legend: { display:false } },
        scales: {
          x: { ticks: { font:{size:11}, color: c.text }, grid: { color: c.grid } },
          y: { beginAtZero:true, ticks: { color: c.text }, grid: { color: c.grid } }
        }
      }
    });
  }

  function renderScoreboard(rows){
    const agg = new Map();
    allCountries.forEach(c => agg.set(c, { enrolled:0, activated:0, paid:0 }));
    rows.forEach(L => {
      const g = agg.get(L.country); if (!g) return;
      g.enrolled++; if (L.activated) g.activated++; if (L.paid) g.paid++;
    });

    let list = [...agg.entries()].map(([country, v]) => ({
      country,
      enrolled: v.enrolled, activated: v.activated, paid: v.paid,
      convEnrolled: v.enrolled ? v.paid/v.enrolled : 0,
      convActivated: v.activated ? v.paid/v.activated : 0
    })).filter(c => c.enrolled > 0);

    const metric = document.getElementById("rankMetric").value;
    list.sort((a,b) => b[metric] - a[metric]);

    const maxEnrolled = Math.max(...list.map(c=>c.enrolled), 1);
    const board = document.getElementById("scoreboard");
    board.innerHTML = "";

    list.forEach((c, i) => {
      const lane = document.createElement("div");
      lane.className = "lane";
      const barWidth = (c.enrolled / maxEnrolled) * 100;
      const actWidth = c.enrolled ? (c.activated / c.enrolled) * 100 : 0;
      const paidWidth = c.enrolled ? (c.paid / c.enrolled) * 100 : 0;

      lane.innerHTML =
        '<div class="rank">'+(i+1)+'</div>' +
        '<div class="lane-country">'+escapeHtml(c.country)+'</div>' +
        '<div class="lane-track-wrap"><div class="track" style="width:'+barWidth.toFixed(1)+'%">' +
          '<div class="seg-activated" style="width:'+actWidth.toFixed(1)+'%"></div>' +
          '<div class="seg-paid" style="width:'+paidWidth.toFixed(1)+'%"></div>' +
        '</div></div>' +
        '<div class="lane-numbers">' +
          '<span>Enrolled <b>'+c.enrolled.toLocaleString()+'</b></span>' +
          '<span>Paid <b>'+c.paid.toLocaleString()+'</b></span>' +
          '<span>Conv <b>'+(c.convEnrolled*100).toFixed(1)+'%</b></span>' +
        '</div>' +
        '<div class="lane-checkbox"><input type="checkbox" title="Add to country filter" '+(filters.countries.has(c.country)?"checked":"")+' /></div>';

      lane.querySelector(".lane-checkbox input").addEventListener("change", (e) => {
        if (e.target.checked) filters.countries.add(c.country); else filters.countries.delete(c.country);
        document.querySelectorAll("#countryChips .chip").forEach(chip => {
          if (chip.textContent === c.country) chip.classList.toggle("active", e.target.checked);
        });
        page = 1; renderAll();
      });

      board.appendChild(lane);
    });
  }

  function renderTable(){
    const rows = getFilteredWithSearch();
    document.getElementById("directoryCount").textContent = rows.length.toLocaleString() + " learners";
    const totalPages = Math.max(1, Math.ceil(rows.length / PAGE_SIZE));
    if (page > totalPages) page = totalPages;
    const startIdx = (page-1)*PAGE_SIZE;
    const pageRows = rows.slice(startIdx, startIdx+PAGE_SIZE);

    const tbody = document.getElementById("tableBody");
    tbody.innerHTML = "";
    pageRows.forEach(L => {
      const tr = document.createElement("tr");

      const progCount = L.programs.length;
      const countBadgeClass = progCount > 1 ? "badge multi" : "badge prog";
      const progCountBadge = '<span class="'+countBadgeClass+'">'+progCount+'</span>';
      const moduleLines = L.programs.map(p => '<span class="module-line">'+escapeHtml(p.program)+': '+escapeHtml(p.course)+' ('+escapeHtml(p.status)+')</span>').join("");

      tr.innerHTML =
        '<td>'+escapeHtml(L.email)+'</td>' +
        '<td>'+escapeHtml((L.first+" "+L.last).trim() || "—")+'</td>' +
        '<td>'+escapeHtml(L.gender)+'</td>' +
        '<td>'+escapeHtml(L.country)+'</td>' +
        '<td>'+progCountBadge+'</td>' +
        '<td>'+moduleLines+'</td>' +
        '<td>'+escapeHtml(L.health)+'</td>' +
        '<td class="'+(L.activated?"yes":"no")+'">'+(L.activated?"Yes":"No")+'</td>' +
        '<td class="'+(L.paid?"yes":"no")+'">'+(L.paid?"Yes":"No")+'</td>';
      tbody.appendChild(tr);
    });

    document.getElementById("pagerInfo").textContent =
      rows.length === 0 ? "No matching learners" :
      "Showing " + (startIdx+1) + "–" + Math.min(startIdx+PAGE_SIZE, rows.length) + " of " + rows.length.toLocaleString();
    document.getElementById("prevPage").disabled = page<=1;
    document.getElementById("nextPage").disabled = page>=totalPages;
  }

  function escapeHtml(str){
    return String(str).replace(/[&<>"']/g, s => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[s]));
  }

})();
